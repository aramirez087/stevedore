### `PreviewCache.swift`

`CachedPreview` is an `NSObject` subclass (required for `NSCache` value type). It stores the
payload and reports its byte cost for `NSCache.totalCostLimit` enforcement.

```swift
// Internal — not public API
final class CachedPreview: NSObject {
    let payload: PreviewPayload
    init(_ payload: PreviewPayload) { self.payload = payload }
    var cost: Int { payload.data.count }
}

public actor PreviewCache {
    private let inner: NSCache<NSString, CachedPreview>

    public init(byteLimit: Int = 50 * 1_024 * 1_024) {
        inner = NSCache<NSString, CachedPreview>()
        inner.totalCostLimit = byteLimit
        inner.countLimit = 0   // unlimited count; cost is the gate
    }

    public func fetch(key: String) -> PreviewPayload? {
        inner.object(forKey: key as NSString)?.payload
    }

    public func store(_ payload: PreviewPayload, forKey key: String) {
        let entry = CachedPreview(payload)
        inner.setObject(entry, forKey: key as NSString, cost: entry.cost)
    }

    public func remove(key: String) {
        inner.removeObject(forKey: key as NSString)
    }

    public func removeAll() {
        inner.removeAllObjects()
    }
}
```

**Why `actor` wrapping `NSCache`:** `NSCache` is an `NSObject` and not `Sendable`. Wrapping in an
`actor` provides the isolation boundary so the cache can be used safely from concurrent async code.

**Byte cap enforcement:** `NSCache.totalCostLimit = byteLimit` causes the cache to automatically
evict least-recently-used entries when total cost exceeds the limit. This is native LRU eviction —
no custom tracking needed. The "burst-1000" test verifies the cap holds after inserting 1,000
entries whose cumulative byte count exceeds the limit.

### `ThumbnailGenerator.swift`

Wraps `QLThumbnailGenerator` (from `QuickLookThumbnailing` framework). Returns PNG `Data`
rather than `NSImage` so the result is `Sendable`.

```swift
import AppKit
import Core
import QuickLookThumbnailing

public actor ThumbnailGenerator {
    // Inflight tasks keyed by "posixPath:WxH"
    private var inflight: [String: Task<Data?, any Error>] = [:]

    public init() {}

    public func thumbnail(for url: URL, size: CGSize) async throws -> Data? {
        let key = inflightKey(url, size)
        if let existing = inflight[key] {
            return try await existing.value
        }
        let task = Task<Data?, any Error> {
            try await Self.generate(url: url, size: size)
        }
        inflight[key] = task
        do {
            let result = try await withTaskCancellationHandler {
                try await task.value
            } onCancel: {
                task.cancel()
            }
            inflight.removeValue(forKey: key)
            return result
        } catch {
            inflight.removeValue(forKey: key)
            throw error
        }
    }

    private func inflightKey(_ url: URL, _ size: CGSize) -> String {
        "\(url.path):\(Int(size.width))x\(Int(size.height))"
    }

    private static func generate(url: URL, size: CGSize) async throws -> Data? {
        let request = QLThumbnailGenerator.Request(
            fileAt: url,
            size: size,
            scale: 1.0,
            representationTypes: .thumbnail
        )
        let generator = QLThumbnailGenerator.shared
        let representation = try await generator.generateBestRepresentation(for: request)
        guard let tiffData = representation.nsImage.tiffRepresentation,
              let bitmap = NSBitmapImageRep(data: tiffData)
        else { return nil }
        return bitmap.representation(using: .png, properties: [:])
    }
}
```

**Coalescing:** The first caller for a given (url, size) creates a `Task` and stores it in
`inflight`. Subsequent concurrent callers for the same key find the task and await its `.value`.
Once the task finishes (success or error), the key is removed so future calls start fresh.

**Cancellation:** `withTaskCancellationHandler` propagates consumer cancellation to the underlying
`Task`. When the consumer's structured task is cancelled, `task.cancel()` is called, which causes
`QLThumbnailGenerator.generateBestRepresentation` to throw `CancellationError` on the next
suspension point.

**`NSImage` → `Data`:** `NSImage` is not `Sendable`. Converting to PNG `Data` before any actor
hop satisfies Swift 6 strict concurrency without `@unchecked Sendable`.

### `QuickLookPanelController.swift`

AppKit interop. All methods must run on the main actor.

```swift
import AppKit
import Core
import QuickLookUI

@MainActor
public final class QuickLookPanelController: NSObject {
    private var previewItems: [NSURL] = []

    public override init() { super.init() }

    /// Open the panel showing `urls`. If already visible, reloads.
    public func show(urls: [URL]) {
        previewItems = urls.map { $0 as NSURL }
        let panel = QLPreviewPanel.shared()
        panel?.dataSource = self
        panel?.delegate = self
        panel?.updateController()
        if panel?.isVisible == true {
            panel?.reloadData()
        } else {
            panel?.makeKeyAndOrderFront(nil)
        }
    }

    /// Toggle panel visibility. First call shows, second call hides.
    public func toggle(urls: [URL]) {
        guard let panel = QLPreviewPanel.shared() else { return }
        if panel.isVisible {
            panel.orderOut(nil)
        } else {
            show(urls: urls)
        }
    }

    /// Close the panel if visible.
    public func close() {
        QLPreviewPanel.shared()?.orderOut(nil)
    }
}

extension QuickLookPanelController: QLPreviewPanelDataSource {
    public nonisolated func numberOfPreviewItems(in panel: QLPreviewPanel!) -> Int {
        // Accessing previewItems on nonisolated is safe: QLPreviewPanel always calls
        // this on main thread; the @MainActor isolation boundary is preserved at runtime.
        MainActor.assumeIsolated { previewItems.count }
    }

    public nonisolated func previewPanel(
        _ panel: QLPreviewPanel!,
        previewItemAt index: Int
    ) -> any QLPreviewItem {
        MainActor.assumeIsolated { previewItems[index] }
    }
}

extension QuickLookPanelController: QLPreviewPanelDelegate {
    public nonisolated func previewPanel(
        _ panel: QLPreviewPanel!,
        sourceFrameOnScreenFor item: any QLPreviewItem
    ) -> NSRect { .zero }
}
```

**`QLPreviewPanelDataSource` is `@objc`:** The protocol methods are called by AppKit on the main
thread. We use `MainActor.assumeIsolated` in the `nonisolated` protocol stubs since we know
AppKit always calls them on main. This avoids `@unchecked Sendable` on `NSURL` items.

**Exit criterion:** Two calls to `toggle(urls:)` in succession open then close the panel cleanly.
Tested by asserting `QLPreviewPanel.shared()?.isVisible` after each toggle.

### `TextPreviewRenderer.swift`

```swift
import Core
import Foundation

public enum TextPreviewRenderer {
    /// Maximum bytes read from file for preview.
    static let readLimit = 1_024 * 1_024   // 1 MB

    public static func render(item: FileItem) async throws -> PreviewPayload? {
        guard item.kind == .regularFile, item.path.scheme == .local else { return nil }
        let url = URL(fileURLWithPath: item.path.posixString)
        return try await Task.detached(priority: .utility) {
            guard let data = try? Data(contentsOf: url, options: .mappedIfSafe) else { return nil }
            let slice = data.prefix(readLimit)
            let (text, encoding) = Self.detectAndDecode(slice)
            guard let text else { return nil }
            _ = encoding   // encoding info available for callers if needed
            guard let rtfData = Self.attributedString(text).rtf(
                from: NSRange(location: 0, length: NSString(string: text).length),
                documentAttributes: [.documentType: NSAttributedString.DocumentType.rtf]
            ) else { return nil }
            return PreviewPayload(mimeType: "text/rtf", data: rtfData)
        }.value
    }

    // Returns (decoded string, detected encoding). Tries BOM first, then UTF-8, then Latin-1.
    static func detectAndDecode(_ data: Data) -> (String?, String.Encoding) {
        // UTF-16 BE BOM: FE FF
        if data.starts(with: [0xFE, 0xFF]) {
            return (String(data: data, encoding: .utf16BigEndian), .utf16BigEndian)
        }
        // UTF-16 LE BOM: FF FE
        if data.starts(with: [0xFF, 0xFE]) {
            return (String(data: data, encoding: .utf16LittleEndian), .utf16LittleEndian)
        }
        // UTF-8 BOM: EF BB BF
        if data.starts(with: [0xEF, 0xBB, 0xBF]) {
            let stripped = data.dropFirst(3)
            return (String(data: stripped, encoding: .utf8), .utf8)
        }
        // Try UTF-8 (most common, no BOM)
        if let s = String(data: data, encoding: .utf8) {
            return (s, .utf8)
        }
        // Fallback: Latin-1 (ISO 8859-1) — always succeeds for any byte sequence
        return (String(data: data, encoding: .isoLatin1), .isoLatin1)
    }

    private static func attributedString(_ text: String) -> NSAttributedString {
        NSAttributedString(
            string: text,
            attributes: [
                .font: NSFont.monospacedSystemFont(ofSize: 12, weight: .regular),
                .foregroundColor: NSColor.textColor,
            ]
        )
    }
}
```

**Off-main-actor:** All file I/O runs inside `Task.detached(priority: .utility)`, so the main
actor is never blocked. `NSAttributedString` and `NSFont` are AppKit types; they are constructed
and consumed within the detached task — never crossing an actor boundary.

### `ImagePreviewRenderer.swift`

```swift
import AppKit
import Core
import Foundation

public enum ImagePreviewRenderer {
    public static func render(item: FileItem, maxDimension: CGFloat = 1024) async throws -> PreviewPayload? {
        guard item.kind == .regularFile, item.path.scheme == .local else { return nil }
        let url = URL(fileURLWithPath: item.path.posixString)
        return try await Task.detached(priority: .utility) {
            guard let image = NSImage(contentsOf: url) else { return nil }
            let resampled = Self.resample(image, maxDimension: maxDimension)
            guard let tiff = resampled.tiffRepresentation,
                  let bitmap = NSBitmapImageRep(data: tiff),
                  let png = bitmap.representation(using: .png, properties: [:])
            else { return nil }
            return PreviewPayload(mimeType: "image/png", data: png)
        }.value
    }

    static func resample(_ image: NSImage, maxDimension: CGFloat) -> NSImage {
        let size = image.size
        guard size.width > 0, size.height > 0 else { return image }
        let scale = min(maxDimension / size.width, maxDimension / size.height, 1.0)
        guard scale < 1.0 else { return image }
        let newSize = NSSize(width: size.width * scale, height: size.height * scale)
        let result = NSImage(size: newSize)
        result.lockFocus()
        image.draw(
            in: NSRect(origin: .zero, size: newSize),
            from: NSRect(origin: .zero, size: size),
            operation: .copy,
            fraction: 1.0
        )
        result.unlockFocus()
        return result
    }
}
```

**`NSImage` isolation:** Construction, drawing, and TIFF conversion all occur within a single
`Task.detached` block. The final `PreviewPayload` contains only `Data` — fully `Sendable`.

**Orientation:** `NSImage` automatically respects EXIF orientation when initialized with
`NSImage(contentsOf:)` on macOS 14+. No manual orientation handling needed.

### `CodePreviewRenderer.swift`

Provides minimal language-aware coloring for ~20 languages using `NSAttributedString` + RTF.
No external parser or dependencies — just regex-based keyword/string/comment highlighting.

```swift
import AppKit
import Core
import Foundation

public enum CodePreviewRenderer {
    public static func render(item: FileItem) async throws -> PreviewPayload? {
        guard item.kind == .regularFile, item.path.scheme == .local else { return nil }
        let url = URL(fileURLWithPath: item.path.posixString)
        let ext = url.pathExtension.lowercased()
        return try await Task.detached(priority: .utility) {
            guard let data = try? Data(contentsOf: url, options: .mappedIfSafe) else { return nil }
            let source = String(data: data.prefix(512 * 1_024), encoding: .utf8)
                ?? String(data: data.prefix(512 * 1_024), encoding: .isoLatin1)
                ?? ""
            let language = Language.detect(extension: ext)
            let attributed = Self.highlight(source: source, language: language)
            guard let rtfData = attributed.rtf(
                from: NSRange(location: 0, length: (source as NSString).length),
                documentAttributes: [.documentType: NSAttributedString.DocumentType.rtf]
            ) else { return nil }
            return PreviewPayload(mimeType: "text/rtf", data: rtfData)
        }.value
    }

    // MARK: - Language enum

    enum Language {
        case swift, python, javascript, typescript, java, go, rust
        case c, cpp, objc, ruby, shell, html, css, json, xml, yaml, sql, unknown

        static func detect(extension ext: String) -> Self {

            case "json": return .json

            case "yaml", "yml": return .yaml
            case "sql": return .sql

            }
        }

        var keywords: [String] {

                        "associatedtype", "some", "any", "subscript", "mutating", "lazy", "weak",
                        "unowned", "defer", "do", "catch"]
            case .python:
                return ["def", "class", "import", "from", "return", "if", "elif", "else", "for",
                        "while", "try", "except", "finally", "with", "as", "pass", "break",
                        "continue", "lambda", "yield", "in", "not", "and", "or", "is", "None",
                        "True", "False", "async", "await", "raise", "del", "global", "nonlocal"]
            case .javascript, .typescript:

    // MARK: - Highlighting

    static func highlight(source: String, language: Language) -> NSAttributedString {
        let result = NSMutableAttributedString(
            string: source,
            attributes: [
                .font: NSFont.monospacedSystemFont(ofSize: 12, weight: .regular),
                .foregroundColor: NSColor.textColor,
            ]
        )
        applyCommentColors(to: result, source: source, language: language)
        applyStringColors(to: result, source: source)
        applyKeywordColors(to: result, source: source, keywords: language.keywords)
        applyNumberColors(to: result, source: source)
        return result
    }

    // Colors: keyword=blue, string=red, comment=green, number=purple
    private static let keywordColor = NSColor.systemBlue
    private static let stringColor = NSColor.systemRed
    private static let commentColor = NSColor.systemGreen
    private static let numberColor = NSColor.systemPurple

    private static func applyKeywordColors(
        to attributed: NSMutableAttributedString,
        source: String,
        keywords: [String]
    ) {
        guard !keywords.isEmpty else { return }
        let pattern = "\\b(" + keywords.map { NSRegularExpression.escapedPattern(for: $0) }
            .joined(separator: "|") + ")\\b"
        applyColor(keywordColor, pattern: pattern, to: attributed, source: source)
    }

    private static func applyStringColors(
        to attributed: NSMutableAttributedString,
        source: String
    ) {
        applyColor(stringColor, pattern: #"\"(?:[^\"\\]|\\.)*\""#, to: attributed, source: source)
        applyColor(stringColor, pattern: #"'(?:[^'\\]|\\.)*'"#, to: attributed, source: source)
    }

    private static func applyCommentColors(
        to attributed: NSMutableAttributedString,
        source: String,
        language: Language
    ) {
        applyColor(commentColor, pattern: "//[^\n]*", to: attributed, source: source)
        applyColor(commentColor, pattern: "#[^\n]*", to: attributed, source: source)
        applyColor(commentColor, pattern: #"/\*[\s\S]*?\*/"#, to: attributed, source: source, options: [])
    }

    private static func applyNumberColors(
        to attributed: NSMutableAttributedString,
        source: String
    ) {
        applyColor(numberColor, pattern: #"\b\d+(\.\d+)?\b"#, to: attributed, source: source)
    }

    private static func applyColor(
        _ color: NSColor,
        pattern: String,
        to attributed: NSMutableAttributedString,
        source: String,
        options: NSRegularExpression.Options = .caseInsensitive
    ) {
        guard let regex = try? NSRegularExpression(pattern: pattern, options: options) else { return }
        let range = NSRange(source.startIndex..., in: source)
        let matches = regex.matches(in: source, range: range)
        for match in matches {
            attributed.addAttribute(.foregroundColor, value: color, range: match.range)
        }
    }
}
```

**No external deps:** All highlighting is pure `NSRegularExpression` + `NSAttributedString`.
The token map is a hardcoded dictionary of keyword lists. `cyclomatic_complexity` is managed
by extracting `Language.keywords` into a `switch` and keeping each `apply*Color` function
single-purpose.

**Fallback:** When `Language.detect` returns `.unknown`, `keywords` is `[]` so `applyKeywordColors`
is a no-op. The output is plain monospaced text — identical to `TextPreviewRenderer` output.

### `PreviewService.swift`

```swift
import AppKit
import Core
import Foundation

public actor PreviewService: PreviewSource {
    private let cache: PreviewCache
    private let thumbnailGenerator: ThumbnailGenerator

    public init(
        cache: PreviewCache = PreviewCache(),
        thumbnailGenerator: ThumbnailGenerator = ThumbnailGenerator()
    ) {
        self.cache = cache
        self.thumbnailGenerator = thumbnailGenerator
    }

    // MARK: - PreviewSource

    public func thumbnail(for item: FileItem, size: CGSize) async throws -> Data? {
        guard item.path.scheme == .local else { return nil }
        let key = "thumb:\(item.path.posixString):\(Int(size.width))x\(Int(size.height))"
        if let cached = await cache.fetch(key: key) {
            return cached.data
        }
        let url = URL(fileURLWithPath: item.path.posixString)
        let data = try await thumbnailGenerator.thumbnail(for: url, size: size)
        if let data {
            await cache.store(PreviewPayload(mimeType: "image/png", data: data), forKey: key)
        }
        return data
    }

    public func preview(for item: FileItem) async throws -> PreviewPayload? {
        guard item.kind == .regularFile, item.path.scheme == .local else { return nil }
        let key = "preview:\(item.path.posixString)"
        if let cached = await cache.fetch(key: key) {
            return cached
        }
        let payload = try await dispatch(item: item)
        if let payload {
            await cache.store(payload, forKey: key)
        }
        return payload
    }

    // MARK: - Renderer dispatch

    private func dispatch(item: FileItem) async throws -> PreviewPayload? {
        let fileName = item.path.lastComponent ?? ""
        let ext = (fileName as NSString).pathExtension.lowercased()

        // 1. Image
        if Self.imageExtensions.contains(ext) {
            return try await ImagePreviewRenderer.render(item: item)
        }

        // 2. Code
        if Self.codeExtensions.contains(ext) {
            return try await CodePreviewRenderer.render(item: item)
        }

        // 3. Text (by extension)
        if Self.textExtensions.contains(ext) {
            return try await TextPreviewRenderer.render(item: item)
        }

        // 4. Magic-byte text detection (binary-safe sniff)
        if await isLikelyText(item: item) {
            return try await TextPreviewRenderer.render(item: item)
        }

        // 5. QuickLook thumbnail fallback
        let url = URL(fileURLWithPath: item.path.posixString)
        let data = try await thumbnailGenerator.thumbnail(for: url, size: CGSize(width: 512, height: 512))
        return data.map { PreviewPayload(mimeType: "image/png", data: $0) }
    }

    private static let imageExtensions: Set<String> = [
        "png", "jpg", "jpeg", "gif", "tiff", "tif", "bmp", "webp", "heic", "heif", "ico", "svg",
    ]

    private static let codeExtensions: Set<String> = [
        "swift", "py", "pyw", "js", "mjs", "ts", "tsx", "java", "go", "rs", "c", "cpp", "cc",
        "cxx", "m", "mm", "rb", "sh", "bash", "zsh", "fish", "html", "htm", "css", "scss",
        "json", "xml", "plist", "yaml", "yml", "toml", "sql", "dockerfile", "makefile",
        "h", "hpp", "command",
    ]

    private static let textExtensions: Set<String> = [
        "txt", "md", "markdown", "rtf", "csv", "log", "conf", "ini", "env",
    ]

    private func isLikelyText(item: FileItem) async -> Bool {
        let url = URL(fileURLWithPath: item.path.posixString)
        return await Task.detached(priority: .utility) {
            guard let data = try? Data(contentsOf: url, options: .mappedIfSafe) else { return false }
            let sample = data.prefix(4_096)
            // A file is likely text if it has no null bytes in the first 4 KB
            return !sample.contains(0x00)
        }.value
    }
}
```

**Actor isolation:** `PreviewService` is an `actor` — its methods run off the main actor executor.
No `@MainActor` annotation; the exit criterion (no main-actor blocking) is enforced structurally.

**Renderer dispatch order:**
1. Image extensions → `ImagePreviewRenderer`
2. Code extensions → `CodePreviewRenderer`
3. Text extensions → `TextPreviewRenderer`
4. Magic bytes (no null bytes in first 4 KB) → `TextPreviewRenderer`
5. Fallback → `ThumbnailGenerator` (QuickLook)

## Key Design Decisions

### 1. `PreviewService` is an `actor`, not `@MainActor`

All preview/thumbnail generation runs on a cooperative thread pool actor, not the main actor.
This is the primary structural guarantee for the exit criterion. No explicit `Thread.isMainThread`
assertions needed — the type system enforces it.

### 2. Renderers are `public enum` namespaces with static `render` methods

Pure namespace enums (no stored state) make the renderers stateless and easily testable in
isolation. They parallel the `RenamePlanner`/`CollisionResolver` pattern from Session 12.

### 3. All data crossing actor boundaries is `Data`

`NSImage` is not `Sendable`. `NSAttributedString` is not `Sendable`. By converting both to
`Data` (PNG and RTF respectively) before returning from `Task.detached` blocks, every
`PreviewPayload` carries only `Data` (which is `Sendable`). No `@unchecked Sendable` needed.

### 4. `PreviewCache` uses `NSCache.totalCostLimit` for byte enforcement

Rather than manually tracking total bytes (which would require an `NSCacheDelegate` to detect
evictions), we set `cache.totalCostLimit = byteLimit` and pass `data.count` as the cost when
inserting. `NSCache` LRU-evicts entries automatically. This is idiomatic and correct; the test
verifies the cap holds under a 1,000-entry burst.

### 5. `ThumbnailGenerator` coalescing via inflight `Task` dictionary

Concurrent requests for the same URL+size reuse the same `Task`. The actor serializes access
to the `inflight` dictionary. This eliminates redundant `QLThumbnailGenerator` calls for
burst-thumbnail scenarios (e.g., scrolling through a large directory).

### 6. `QuickLookPanelController` uses `MainActor.assumeIsolated` for protocol stubs

`QLPreviewPanelDataSource`/`QLPreviewPanelDelegate` are `@objc` protocols whose methods
AppKit always calls on the main thread. Using `MainActor.assumeIsolated` in `nonisolated`
stubs avoids `@unchecked Sendable` while satisfying Swift 6's isolation rules.

### 7. Magic-byte text detection: null-byte sniff

Rather than implementing a full MIME-type sniffer, we use a pragmatic heuristic: if the first
4 KB of a file contains no null bytes (`0x00`), it is treated as text. This correctly identifies
UTF-8, ASCII, and Latin-1 text files. Binary files (ELF, Mach-O, JPEG, ZIP) contain null bytes
early and fall through to the QuickLook thumbnail.

### 8. `QLThumbnailGenerator` API: use `generateBestRepresentation(for:)` with `async`

On macOS 14, `QLThumbnailGenerator` provides `generateBestRepresentation(for:)` as an `async`
function (no `withCheckedContinuation` wrapping needed). We use it directly inside a
`Task.detached(priority: .utility)` to keep the actor executor free.

### 9. Package.swift not modified

All frameworks used (`QuickLookThumbnailing`, `QuickLookUI`, `AppKit`) are macOS system frameworks
available without SPM packages. `FeaturesPreview` target retains its original `extraDependencies: []`.

## Risks and Mitigations

| Risk | Mitigation |
|---|---|
| `NSAttributedString.rtf(from:documentAttributes:)` returns `nil` on some content | Return `nil` from renderer; let `PreviewService.dispatch` fall through to QuickLook |
| `QLThumbnailGenerator.generateBestRepresentation` is `async` only on macOS 13+ | macOS 14 deployment target — safe |
| `QLPreviewPanel.shared()` returns `Optional` | Guard with `guard let panel = QLPreviewPanel.shared()` |
| `NSCache` eviction is non-deterministic | Test uses a small `byteLimit` relative to payload sizes to force eviction; asserts not all 1,000 entries are live |
| `NSImage(contentsOf:)` can block if file is on a slow volume | Wrapped in `Task.detached(priority: .utility)`; no main-actor blocking |
| SwiftLint `nesting` rule: `Language` enum inside `CodePreviewRenderer` enum | `Language` is a module-internal `enum` (no access modifier) at file scope, not nested inside `CodePreviewRenderer` |
| SwiftLint `cyclomatic_complexity` for `Language.keywords` switch (20+ cases) | The switch has 20 cases but each case is a single `return` statement — complexity is 20, at the warning threshold (12). Extract to a `var` on `Language` to keep cyclomatic complexity per function at 1 |
| `swiftformat --lint` paths-before-flag | Per cerebrum: `swiftformat Sources/Features/Preview Tests/FeaturesTests/PreviewTests --lint` |
| `prefer_self_in_static_references` lint rule | All static arrays in `PreviewService` use `Self.imageExtensions` etc. |
| `implicit_return` lint rule | Single-expression closures and computed properties use implicit return |
| `sorted_imports` lint rule | Always alphabetize: `AppKit`, `Core`, `Foundation`, `QuickLookThumbnailing`, `QuickLookUI`, `XCTest` |
| File basename collision in `FeaturesTests` | All 7 new test filenames verified unique against existing `OperationsTests/`, `SyncTests/`, `RenameTests/` |
| `QLPreviewPanelDataSource.numberOfPreviewItems` is not `nonisolated` in `@MainActor` class | Use `@MainActor final class` + `nonisolated` stubs with `MainActor.assumeIsolated`; document the invariant |
| `NSRegularExpression` multiline patterns for block comments | Use `.dotMatchesLineSeparators` option for `/* ... */` patterns |

## Notes for Session Implementer

- **`QLThumbnailGenerator` is available from `QuickLookThumbnailing`** — import
  `QuickLookThumbnailing` not `QuickLookUI`. The panel controller imports `QuickLookUI`.
  Keep these imports in separate files to avoid confusion.

- **`QLPreviewPanel.shared()` returns `QLPreviewPanel?`** — always guard. On headless CI
  (no window server), this can return `nil`.

- **`NSAttributedString.rtf(from:documentAttributes:)` signature:** The `from:` parameter is
  `NSRange` (Objective-C range), not `Range<String.Index>`. Use
  `NSRange(location: 0, length: (source as NSString).length)`.

- **`Language.keywords` switch in `CodePreviewRenderer`** — SwiftLint `cyclomatic_complexity`
  counts each `case` in a switch. With 20+ languages, the `keywords` computed property will
  exceed the warning threshold of 12. Extract to a dedicated `func keywords(for language: Language) -> [String]`
  function so the switch is contained in that function (one complexity count per function).

- **The `FeaturesTests` test target spans all directories under `Tests/FeaturesTests/`.**
  The new `PreviewTests/` subdirectory is automatically included. Verify no basename
  collision with existing files: `PreviewTestSupport.swift` is unique; all 7 new filenames
  are verified unique against `OperationsTests/`, `SyncTests/`, `RenameTests/`.

- **`swiftformat` 0.59 `--lint` flag:** Per cerebrum, pass paths before flag:
  `swiftformat Sources/Features/Preview Tests/FeaturesTests/PreviewTests --lint`.

- **`NSCache` and `@Sendable` closures:** `NSCache` is not `Sendable`, but it's only ever
  accessed through the actor's synchronized methods. No closure captures of `NSCache` across
  actor boundaries. The `CachedPreview` class inherits from `NSObject` and is not `Sendable` —
  it never leaves the actor either.

- **`QuickLookPanelController` headless test guard:** `QLPreviewPanel.shared()` may return
  `nil` in CI environments without a display. The toggle test should skip if
  `QLPreviewPanel.shared() == nil`.
