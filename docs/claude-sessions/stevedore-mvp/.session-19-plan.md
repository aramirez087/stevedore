### HistoryStack.swift

```swift
public struct HistoryStack: Sendable {
    public static let defaultCapacity = 64

    private let capacity: Int
    private var back: [FilePath]     // oldest→newest; pop from end for goBack
    private var forward: [FilePath]  // next first; remove from front for goForward
    public private(set) var current: FilePath

    public init(root: FilePath, capacity: Int = defaultCapacity)

    public var canGoBack: Bool { !back.isEmpty }
    public var canGoForward: Bool { !forward.isEmpty }

    // Returns false when path == current (no-op). Clears forward. Caps back.
    @discardableResult
    public mutating func push(_ path: FilePath) -> Bool

    // Returns false at oldest entry (no-op). Moves current → front of forward.
    @discardableResult
    public mutating func goBack() -> Bool

    // Returns false when forward empty (no-op). Moves current → back.
    @discardableResult
    public mutating func goForward() -> Bool
}
```

**Capacity invariant**: `back.count` never exceeds `capacity`. On each push or goForward that appends to `back`, if `back.count > capacity`, `back.removeFirst()`.

**push logic**:
1. Guard `path != current`, else return false.
2. `forward.removeAll()`
3. `back.append(current)`; if `back.count > capacity`: `back.removeFirst()`
4. `current = path`; return true

**goBack logic**:
1. Guard `!back.isEmpty`, else return false.
2. `forward.insert(current, at: 0)`
3. `current = back.removeLast()`; return true

**goForward logic**:
1. Guard `!forward.isEmpty`, else return false.
2. `back.append(current)`; if `back.count > capacity`: `back.removeFirst()`
3. `current = forward.removeFirst()`; return true

### PaneToolbarViewModel.swift

```swift
import Core
import Observation
import SwiftUI

@Observable
@MainActor
public final class PaneToolbarViewModel {
    // Published state (observed by SwiftUI)
    public private(set) var currentPath: FilePath
    public private(set) var canGoBack: Bool = false
    public private(set) var canGoForward: Bool = false
    public private(set) var viewMode: ViewMode = .list
    public private(set) var searchText: String = ""

    // Callbacks — set by the shell before embedding the view
    public var onNavigate: (FilePath) -> Void = { _ in }
    public var onRefresh: () -> Void = {}
    public var onNewFolder: () -> Void = {}
    public var onViewModeUnavailable: (ViewMode) -> Void = { _ in }

    private var history: HistoryStack

    public init(root: FilePath, capacity: Int = HistoryStack.defaultCapacity) {
        self.history = HistoryStack(root: root, capacity: capacity)
        self.currentPath = root
    }

    // Called by the shell when external navigation occurs (e.g., folder double-click).
    // Pushes to history WITHOUT firing onNavigate (avoids feedback loop).
    public func navigate(to path: FilePath) {
        history.push(path)
        sync()
    }

    public func goBack() {
        guard history.goBack() else { return }
        sync()
        onNavigate(history.current)
    }

    public func goForward() {
        guard history.goForward() else { return }
        sync()
        onNavigate(history.current)
    }

    public func goUp() {
        guard let parent = history.current.parent else { return }
        history.push(parent)
        sync()
        onNavigate(parent)
    }

    public func refresh() { onRefresh() }
    public func newFolder() { onNewFolder() }

    public func setViewMode(_ mode: ViewMode) {
        if mode == .list {
            viewMode = mode
        } else {
            onViewModeUnavailable(mode)
        }
    }

    public func setSearch(_ term: String) {
        searchText = term
    }

    // Keeps published properties in sync with HistoryStack
    private func sync() {
        currentPath = history.current
        canGoBack = history.canGoBack
        canGoForward = history.canGoForward
    }
}
```

**Key design decision**: `navigate(to:)` vs action methods. `navigate(to:)` is called *by the shell* to record external navigation without echoing. `goBack`, `goForward`, `goUp` are called by *the toolbar buttons* and both update history AND fire `onNavigate` to drive the shell.

### SearchDebouncer.swift

A `@MainActor` class that encapsulates debounce logic and is testable in isolation.

```swift
import Foundation

@MainActor
public final class SearchDebouncer {
    public var onSearch: ((String) -> Void)?

    private let delay: Duration
    private let sleep: @Sendable (Duration) async throws -> Void
    private var pendingTask: Task<Void, Never>?

    public init(
        delay: Duration = .milliseconds(250),
        sleep: @escaping @Sendable (Duration) async throws -> Void =
            { try await Task.sleep(for: $0) }
    ) {
        self.delay = delay
        self.sleep = sleep
    }

    public func schedule(_ term: String) {
        pendingTask?.cancel()
        let sleepFn = self.sleep
        let delay = self.delay
        pendingTask = Task { @MainActor [weak self] in
            try? await sleepFn(delay)
            guard !Task.isCancelled, let self else { return }
            self.onSearch?(term)
        }
    }

    public func clear() {
        pendingTask?.cancel()
        pendingTask = nil
        onSearch?("")
    }
}
```

**Why a class**: mutable `pendingTask` state that outlives individual schedule calls. `[weak self]` avoids retain cycle when the debouncer is stored as a `@State` object or inside a VM.

### PathBarModel.swift

Pure value type that derives breadcrumb segments from a `FilePath`. Kept `internal` so tests can use `@testable import UIToolbar`.

```swift
import Core

struct PathBarSegmentInfo: Equatable {
    let path: FilePath
    let label: String
    let isLast: Bool
}

enum PathBarItem: Equatable {
    case segment(PathBarSegmentInfo)
    case ellipsis(collapsed: [PathBarSegmentInfo])
}

struct PathBarModel {
    static let maxVisible = 5

    // Build all segments for a path (root + each component)
    static func allSegments(for path: FilePath) -> [PathBarSegmentInfo] {
        var result: [PathBarSegmentInfo] = []
        let root = FilePath.root(path.scheme)
        let rootLabel: String = path.scheme == .local ? "/" : "\(path.scheme.rawValue)://"
        result.append(PathBarSegmentInfo(path: root, label: rootLabel, isLast: path.components.isEmpty))

        for i in path.components.indices {
            let subPath = FilePath(scheme: path.scheme, components: Array(path.components.prefix(i + 1)))
            let isLast = i == path.components.indices.last
            result.append(PathBarSegmentInfo(path: subPath, label: path.components[i], isLast: isLast))
        }
        return result
    }

    // Returns visible items: may include an .ellipsis node for overflow
    static func visibleItems(for path: FilePath) -> [PathBarItem] {
        let all = allSegments(for: path)
        guard all.count > maxVisible else {
            return all.map { .segment($0) }
        }
        // Show first 2 + ellipsis + last 2 (adjust isLast only on true last)
        let head = Array(all.prefix(2))
        let tail = Array(all.suffix(2))
        let collapsed = Array(all.dropFirst(2).dropLast(2))
        var items: [PathBarItem] = head.map { .segment($0) }
        items.append(.ellipsis(collapsed: collapsed))
        items += tail.map { .segment($0) }
        return items
    }
}
```

**Why separate from PathBar**: breadcrumb derivation is pure logic; testing it with `@testable import` avoids SwiftUI rendering overhead for the logic tests.

### PathBarSegment.swift

```swift
import Core
import DesignSystem
import SwiftUI

public struct PathBarSegment: View {
    private let label: String
    private let isLast: Bool
    private let onTap: () -> Void
    private let loadChildren: (@Sendable () async -> [String])?

    @State private var showPopover = false
    @State private var children: [String] = []
    @State private var isLoading = false
    @Environment(\.theme) private var theme

    public init(
        label: String,
        isLast: Bool = false,
        onTap: @escaping () -> Void,
        loadChildren: (@Sendable () async -> [String])? = nil
    ) { ... }

    public var body: some View {
        HStack(spacing: 2) {
            Button(action: onTap) {
                Text(label)
                    .font(isLast
                        ? theme.typography.body.weight(.semibold)
                        : theme.typography.body)
                    .foregroundStyle(isLast
                        ? theme.colors.textPrimary
                        : theme.colors.textSecondary)
                    .lineLimit(1)
            }
            .buttonStyle(.plain)

            if !isLast {
                Image(systemName: "chevron.right")
                    .font(.caption2)
                    .foregroundStyle(theme.colors.textSecondary)
                    .contentShape(Rectangle())
                    .onTapGesture {
                        guard loadChildren != nil else { return }
                        showPopover = true
                    }
                    .popover(isPresented: $showPopover) {
                        childrenPopover
                    }
            }
        }
    }

    @ViewBuilder
    private var childrenPopover: some View {
        Group {
            if isLoading {
                ProgressView().padding(Spacing.md)
            } else {
                ScrollView {
                    VStack(alignment: .leading, spacing: 0) {
                        ForEach(children, id: \.self) { name in
                            Button(name) {
                                showPopover = false
                                onTap()  // shell interprets as navigation intent
                            }
                            .buttonStyle(.plain)
                            .padding(.horizontal, Spacing.md)
                            .padding(.vertical, Spacing.sm)
                        }
                    }
                }
                .frame(minWidth: 200, maxHeight: 300)
            }
        }
        .task {
            guard let loader = loadChildren else { return }
            isLoading = true
            children = await loader()
            isLoading = false
        }
    }
}
```

**Exit criterion compliance**: `loadChildren` is only called inside `.task` when the popover is shown, never at construction time.

### PathBar.swift

```swift
import Core
import DesignSystem
import SwiftUI

public struct PathBar: View {
    private let path: FilePath
    private let onNavigate: (FilePath) -> Void
    private let loadChildren: (FilePath) async -> [String]

    @Environment(\.theme) private var theme

    public init(
        path: FilePath,
        onNavigate: @escaping (FilePath) -> Void,
        loadChildren: @escaping (FilePath) async -> [String] = { _ in [] }
    ) { ... }

    public var body: some View {
        ScrollView(.horizontal, showsIndicators: false) {
            HStack(spacing: 0) {
                ForEach(PathBarModel.visibleItems(for: path), id: \.id) { item in
                    switch item {
                    case .segment(let info):
                        PathBarSegment(
                            label: info.label,
                            isLast: info.isLast,
                            onTap: { onNavigate(info.path) },
                            loadChildren: info.isLast ? nil : {
                                await loadChildren(info.path)
                            }
                        )
                    case .ellipsis(let collapsed):
                        EllipsisMenu(items: collapsed, onNavigate: onNavigate)
                    }
                }
            }
            .padding(.horizontal, Spacing.sm)
        }
        .frame(maxWidth: .infinity)
    }
}
```

`PathBarItem` needs `Identifiable` conformance for `ForEach`. Add computed `id` property or a stable ID using the path's `posixString`.

**EllipsisMenu** is a private helper view inside `PathBar.swift` (tightly coupled):

            }
        } label: {

                .font(.body)
        }
        .menuStyle(.borderlessButton)

}

### PaneToolbar.swift

```swift
import Core
import DesignSystem
import SwiftUI

public struct PaneToolbar: View {
    @Bindable private var viewModel: PaneToolbarViewModel
    private let loadChildren: (FilePath) async -> [String]

    @State private var searchText = ""
    @State private var debouncer: SearchDebouncer

    @Environment(\.theme) private var theme

    public init(
        viewModel: PaneToolbarViewModel,
        loadChildren: @escaping (FilePath) async -> [String] = { _ in [] }
    ) {
        self.viewModel = viewModel
        self.loadChildren = loadChildren
        self._debouncer = State(initialValue: SearchDebouncer())
    }

    public var body: some View {
        HStack(spacing: Spacing.sm) {
            navButtons
            Divider().frame(height: 20)
            PathBar(
                path: viewModel.currentPath,
                onNavigate: { viewModel.navigate(to: $0) },
                loadChildren: loadChildren
            )
            .layoutPriority(1)
            Divider().frame(height: 20)
            viewModePicker
            Divider().frame(height: 20)
            SearchField(text: $searchText, debouncer: debouncer)
                .frame(minWidth: 120, maxWidth: 200)
            actionButtons
        }
        .padding(.horizontal, Spacing.sm)
        .padding(.vertical, Spacing.xs)
        .background(theme.colors.surface)
        .onAppear {
            debouncer.onSearch = { [weak viewModel] term in
                viewModel?.setSearch(term)
            }
        }
        .onChange(of: searchText) { _, new in
            if new.isEmpty { viewModel.setSearch("") }
        }
    }

    private var navButtons: some View {
        HStack(spacing: Spacing.xs) {
            Button { viewModel.goBack() } label: {
                Image(systemName: "chevron.left")
            }
            .disabled(!viewModel.canGoBack)
            .buttonStyle(.plain)

            Button { viewModel.goForward() } label: {
                Image(systemName: "chevron.right")
            }
            .disabled(!viewModel.canGoForward)
            .buttonStyle(.plain)

            Button { viewModel.goUp() } label: {
                Image(systemName: "arrow.up")
            }
            .disabled(viewModel.currentPath.isRoot)
            .buttonStyle(.plain)
        }
    }

    private var viewModePicker: some View {
        HStack(spacing: 2) {
            ForEach(ViewMode.allCases, id: \.self) { mode in
                Button { viewModel.setViewMode(mode) } label: {
                    Image(systemName: iconName(for: mode))
                        .foregroundStyle(
                            viewModel.viewMode == mode
                                ? theme.colors.accent
                                : theme.colors.textSecondary
                        )
                }
                .buttonStyle(.plain)
            }
        }
    }

    private var actionButtons: some View {
        HStack(spacing: Spacing.xs) {
            Button { viewModel.refresh() } label: {
                Image(systemName: "arrow.clockwise")
            }
            .buttonStyle(.plain)

            Button { viewModel.newFolder() } label: {
                Image(systemName: "folder.badge.plus")
            }
            .buttonStyle(.plain)
        }
    }

    private func iconName(for mode: ViewMode) -> String {
        switch mode {
        case .list: "list.bullet"
        case .columns: "rectangle.split.3x1"
        case .icons: "square.grid.2x2"
        }
    }
}
```

`@Bindable` requires the type to be `@Observable`. `PaneToolbarViewModel` is `@Observable`, so `@Bindable var viewModel: PaneToolbarViewModel` works in SwiftUI on macOS 14+.

## Key Design Decisions

1. **`HistoryStack` is a value type (`struct`)** — fits the "all public types `Sendable`" requirement naturally; no actor isolation needed; VM owns it as a private var.

2. **`PaneToolbarViewModel` is `@MainActor @Observable class`** — `@Observable` synthesis requires a class (reference type) for the observation tracking mechanism. `@MainActor` matches SwiftUI's main-thread rendering model. `@Bindable` in `PaneToolbar` requires this.

3. **Separate `SearchDebouncer` from `SearchField`** — Debounce logic is independently testable without going through SwiftUI's `onChange` modifier (which only fires in a live view update cycle). This is the only way to write a deterministic debounce test.

4. **`PathBarModel` as an `internal` struct** — All breadcrumb segment derivation lives here. Tests use `@testable import UIToolbar` to access it. The View (`PathBar`) delegates to `PathBarModel`; its `body` only does layout.

5. **`loadChildren` callback on `PathBar`/`PathBarSegment`** — The popover content is loaded only when the user opens it (via `.task` modifier). This satisfies the zero-FS-access-at-construction requirement.

6. **Ellipsis strategy: first 2 + ellipsis + last 2** — Always shows the scheme root and at least one ancestor, always shows the final component. Collapsed middle items appear in a `Menu` (native macOS popover-like dropdown). Threshold: > 5 total segments.

7. **Overflow via `ScrollView` (secondary)** — `PathBar` wraps in a horizontal `ScrollView` as a fallback for very long paths before the ellipsis kicks in. This is defensive — in practice the ellipsis threshold of 5 covers most paths.

8. **No `@unchecked Sendable`** — `PaneToolbarViewModel` is `@MainActor`, which is safer than `@unchecked Sendable`. `HistoryStack` is a `struct` and therefore `Sendable` by default (all stored types are `Sendable`). `SearchDebouncer` is `@MainActor` (not `Sendable`).

9. **View mode "not yet implemented" is handled in the VM**, not the view — the VM's `setViewMode` fires `onViewModeUnavailable` for `.columns` and `.icons`. The shell receives this and can show a banner/alert. The toolbar view itself is stateless about this.

10. **`PaneToolbar` uses `@Bindable` (not `@ObservedObject`)** — `@Bindable` is the correct SwiftUI binding for `@Observable` types on macOS 14+. `@ObservedObject` requires `ObservableObject` conformance.

## Risks and Mitigations

| Risk | Mitigation |
|---|---|
| `@Observable` + `@MainActor` strict concurrency errors in Swift 6 | Mark all callback properties as plain `(T) -> Void`; they're called within `@MainActor` isolation already |
| `PathBarItem` needs `Identifiable` for `ForEach` | Use `\.id` computed property; for `.segment` use `info.path.posixString`; for `.ellipsis` use a stable constant string |
| `swiftformat --lint` argument order bug | Use paths-first order: `swiftformat Sources/UI/Toolbar Tests/UITests/ToolbarTests --lint` |
| `@Bindable` requires explicit import of Observation framework | `import SwiftUI` is sufficient on macOS 14+; Observation is re-exported |
