### `HistoryStack.swift`

```swift
import Core

public struct HistoryStack: Sendable {
    public static let defaultCapacity: Int = 64

    public let capacity: Int
    private var backStack: [FilePath]    // oldest → newest; .last is current
    private var forwardStack: [FilePath] // .last is next forward item

    public init(capacity: Int = defaultCapacity) {
        self.capacity = capacity
        self.backStack = []
        self.forwardStack = []
    }

    public var current: FilePath? { backStack.last }
    public var canGoBack: Bool { backStack.count > 1 }
    public var canGoForward: Bool { !forwardStack.isEmpty }

    public mutating func navigate(to path: FilePath) {
        forwardStack.removeAll()
        backStack.append(path)
        if backStack.count > capacity { backStack.removeFirst() }
    }

    @discardableResult
    public mutating func goBack() -> FilePath? {
        guard canGoBack else { return nil }
        let popped = backStack.removeLast()
        forwardStack.append(popped)
        return backStack.last
    }

    @discardableResult
    public mutating func goForward() -> FilePath? {
        guard canGoForward else { return nil }
        let next = forwardStack.removeLast()
        backStack.append(next)
        return next
    }
}
```

**Design:** `backStack.last` is the current path — no separate `current` field needed.
`canGoBack` requires `count > 1` (not `> 0`) because position 0 is the oldest visited location
with nowhere further to go back. `PaneToolbarViewModel.init` calls `history.navigate(to:
initialPath)` to seed the first entry, so `canGoBack` starts false.

**Capacity:** applied in `navigate(to:)` only. `goBack`/`goForward` shuffle items between stacks
without adding new ones, so `backStack.count + forwardStack.count ≤ capacity` always holds after
the initial `navigate` call caps the back stack.

### `SearchDebouncer.swift`

```swift
import Foundation
import Observation

public typealias SleepFunction = @Sendable (Duration) async throws -> Void

@MainActor
@Observable
public final class SearchDebouncer {
    public private(set) var term: String = ""
    public var onFire: ((String) -> Void)?

    private let interval: Duration
    private let sleep: SleepFunction
    private var pendingTask: Task<Void, Never>?

    public init(
        interval: Duration = .milliseconds(250),
        sleep: @escaping SleepFunction = { try await Task.sleep(for: $0) }
    ) {
        self.interval = interval
        self.sleep = sleep
    }

    public func update(_ newTerm: String) {
        term = newTerm
        pendingTask?.cancel()
        let snapshot = newTerm
        let sleepFn = sleep
        let fireInterval = interval
        pendingTask = Task { [weak self] in
            try? await sleepFn(fireInterval)
            guard !Task.isCancelled, let self else { return }
            self.onFire?(snapshot)
        }
    }

    public func clear() {
        pendingTask?.cancel()
        pendingTask = nil
        term = ""
        onFire?("")
    }
}
```

**Concurrency notes:**
- Class is `@MainActor`, so all property access is main-actor-isolated.
- `Task { }` created from `@MainActor` context inherits main actor — the task runs on the main
  actor executor, suspending it during `sleepFn(...)` to allow other work.
- The `sleep` closure is captured by copy and is `@Sendable`, preventing data races.
- `[weak self]` avoids retain cycles; the `guard let self` after sleep is required.

**Testing:** inject `sleep: { _ in }` (no-op closure, returns immediately). With zero delay, the
only non-cancelled Task will fire after two `await Task.yield()` calls in the test.

`Duration` is a Swift stdlib type (Swift 5.7+, macOS 13+). `Task.sleep(for:)` is macOS 13+.
Both are available on our macOS 14 deployment target.

### `PaneToolbarViewModel.swift`

```swift
import Core
import Observation

@MainActor
@Observable
public final class PaneToolbarViewModel {
    public private(set) var currentPath: FilePath
    public var viewMode: ViewMode = .list
    public private(set) var history: HistoryStack
    public let searchDebouncer: SearchDebouncer

    public var onNavigate: ((FilePath) -> Void)?
    public var onRefresh: (() -> Void)?
    public var onNewFolder: (() -> Void)?
    public var onViewModeUnavailable: ((ViewMode) -> Void)?

    public var canGoBack: Bool { history.canGoBack }
    public var canGoForward: Bool { history.canGoForward }
    public var canGoUp: Bool { !currentPath.isRoot }

    public init(
        initialPath: FilePath,
        historyCapacity: Int = HistoryStack.defaultCapacity,
        searchDebouncer: SearchDebouncer? = nil
    ) {
        self.currentPath = initialPath
        self.history = HistoryStack(capacity: historyCapacity)
        self.searchDebouncer = searchDebouncer ?? SearchDebouncer()
        self.history.navigate(to: initialPath)
    }

    public func navigate(to path: FilePath) {
        history.navigate(to: path)
        currentPath = path
        onNavigate?(path)
    }

    public func goBack() {
        guard let path = history.goBack() else { return }
        currentPath = path
        onNavigate?(path)
    }

    public func goForward() {
        guard let path = history.goForward() else { return }
        currentPath = path
        onNavigate?(path)
    }

    public func goUp() {
        guard let parent = currentPath.parent else { return }
        navigate(to: parent)
    }

    public func refresh() { onRefresh?() }
    public func newFolder() { onNewFolder?() }

    public func setViewMode(_ mode: ViewMode) {
        if mode == .list {
            viewMode = mode
        } else {
            onViewModeUnavailable?(mode)
        }
    }
}
```

**Key contracts:**
- `searchDebouncer` is public so the shell can wire `searchDebouncer.onFire`.
- `history` is public (read) so tests and `@testable` callers can inspect state.
- `setViewMode(.list)` sets `viewMode` and does NOT call `onViewModeUnavailable`.
- `setViewMode(.columns/.icons)` calls `onViewModeUnavailable` and does NOT change `viewMode`.

### `PathBarSegment.swift`

**SwiftLint `nesting` rule warning:** The rule disallows types nested beyond 1 level. If
`SegmentData` were declared inside `BreadcrumbItem`, that would be nesting level 2 (a struct
inside an enum inside a file) — **this fails `swiftlint --strict`**. Both supporting types must
be at file scope.

```swift
import Core
import DesignSystem
import SwiftUI

// Internal — only PathBar constructs these.
struct PathSegmentData {
    let path: FilePath
    let label: String
    let isLast: Bool
}

enum BreadcrumbItem: Identifiable {
    case segment(id: Int, data: PathSegmentData)
    case ellipsis(id: Int, hidden: [FilePath])

    var id: Int {
        switch self {
        case let .segment(id, _): id
        case let .ellipsis(id, _): id
        }
    }
}

public struct PathBarSegment: View {
    public let path: FilePath
    public let label: String
    public let isLast: Bool
    public var onTap: (FilePath) -> Void
    public var subfolderProvider: @MainActor (FilePath) async -> [FilePath]

    @State private var isShowingPicker = false
    @State private var subfolders: [FilePath] = []
    @State private var isLoading = false

    @Environment(\.theme) private var theme

    public init(
        path: FilePath,
        label: String,
        isLast: Bool,
        onTap: @escaping (FilePath) -> Void,
        subfolderProvider: @escaping @MainActor (FilePath) async -> [FilePath] = { _ in [] }
    ) {
        self.path = path
        self.label = label
        self.isLast = isLast
        self.onTap = onTap
        self.subfolderProvider = subfolderProvider
    }

    public var body: some View {
        HStack(spacing: Spacing.xs) {
            Button(label) { onTap(path) }
                .buttonStyle(.plain)
                .font(theme.typography.body)
                .foregroundStyle(isLast ? theme.colors.textPrimary : theme.colors.textSecondary)

            if !isLast {
                Button {
                    isShowingPicker = true
                } label: {
                    Image(systemName: "chevron.right")
                        .foregroundStyle(theme.colors.textSecondary)
                        .imageScale(.small)
                }
                .buttonStyle(.plain)
                .popover(isPresented: $isShowingPicker) {
                    subfolderPickerContent
                }
                .task(id: isShowingPicker) {
                    guard isShowingPicker else { return }
                    isLoading = true
                    subfolders = await subfolderProvider(path)
                    isLoading = false
                }
            }
        }
    }

    @ViewBuilder
    private var subfolderPickerContent: some View {
        Group {
            if isLoading {
                ProgressView()
                    .padding(Spacing.md)
            } else if subfolders.isEmpty {
                Text("No subfolders")
                    .font(theme.typography.caption)
                    .foregroundStyle(theme.colors.textSecondary)
                    .padding(Spacing.md)
            } else {
                VStack(alignment: .leading, spacing: 0) {
                    ForEach(subfolders, id: \.self) { sub in
                        Button(sub.displayName) {
                            isShowingPicker = false
                            onTap(sub)
                        }
                        .buttonStyle(.plain)
                        .font(theme.typography.body)
                        .padding(.horizontal, Spacing.sm)
                        .padding(.vertical, Spacing.xs)
                    }
                }
                .padding(.vertical, Spacing.xs)
            }
        }
        .frame(minWidth: 160)
    }
}
```

**Zero FS access at construction time:** `subfolderProvider` is never called in `init` or
during view layout; only fired when the chevron popover opens (via `.task(id: isShowingPicker)`).

### `PathBar.swift`

```swift
import Core
import DesignSystem
import SwiftUI

public struct PathBar: View {
    public let path: FilePath
    public var onNavigate: (FilePath) -> Void
    public var subfolderProvider: @MainActor (FilePath) async -> [FilePath]

    // Collapse middle segments when total exceeds this threshold.
    static let maxVisible: Int = 5

    @Environment(\.theme) private var theme

    public init(
        path: FilePath,
        onNavigate: @escaping (FilePath) -> Void,
        subfolderProvider: @escaping @MainActor (FilePath) async -> [FilePath] = { _ in [] }
    ) {
        self.path = path
        self.onNavigate = onNavigate
        self.subfolderProvider = subfolderProvider
    }

    public var body: some View {
        ScrollView(.horizontal, showsIndicators: false) {
            HStack(spacing: 0) {
                ForEach(breadcrumbs) { item in
                    switch item {
                    case let .segment(_, data):
                        PathBarSegment(
                            path: data.path,
                            label: data.label,
                            isLast: data.isLast,
                            onTap: onNavigate,
                            subfolderProvider: subfolderProvider
                        )
                    case let .ellipsis(_, hidden):
                        ellipsisButton(hidden: hidden)
                    }
                }
            }
            .padding(.horizontal, Spacing.sm)
        }
        .frame(height: 28)
        .background(theme.colors.surface)
    }

    // Internal — accessible via @testable import for breadcrumb-derivation tests.
    var breadcrumbs: [BreadcrumbItem] {
        var prefixes: [(path: FilePath, label: String)] = []
        let root = FilePath.root(path.scheme)
        prefixes.append((root, root.displayName))
        for i in 0..<path.components.count {
            let sub = FilePath(scheme: path.scheme, components: Array(path.components.prefix(i + 1)))
            prefixes.append((sub, path.components[i]))
        }

        guard prefixes.count > Self.maxVisible else {
            return prefixes.enumerated().map { idx, pair in
                .segment(
                    id: idx,
                    data: PathSegmentData(path: pair.path, label: pair.label, isLast: idx == prefixes.count - 1)
                )
            }
        }

        // Overflow: keep first 1 (root) + ellipsis + last (maxVisible - 2).
        let tailCount = Self.maxVisible - 2
        let head = prefixes[0]
        let tail = Array(prefixes.suffix(tailCount))
        let hidden = Array(prefixes[1..<(prefixes.count - tailCount)])

        var items: [BreadcrumbItem] = []
        items.append(.segment(id: 0, data: PathSegmentData(path: head.path, label: head.label, isLast: false)))
        items.append(.ellipsis(id: 1, hidden: hidden.map(\.path)))
        for (i, pair) in tail.enumerated() {
            let isLast = i == tail.count - 1
            items.append(.segment(id: i + 2, data: PathSegmentData(path: pair.path, label: pair.label, isLast: isLast)))
        }
        return items
    }

    private func ellipsisButton(hidden: [FilePath]) -> some View {
        Menu {
            ForEach(hidden, id: \.self) { p in
                Button(p.displayName) { onNavigate(p) }
            }
        } label: {
            HStack(spacing: Spacing.xs) {
                Text("…")
                    .foregroundStyle(theme.colors.textSecondary)
                    .font(theme.typography.body)
                Image(systemName: "chevron.right")
                    .foregroundStyle(theme.colors.textSecondary)
                    .imageScale(.small)
            }
        }
        .buttonStyle(.plain)
    }
}
```

**Overflow algorithm:** with `maxVisible = 5`, a 7-component path (`/a/b/c/d/e/f/g`) has 8
prefixes. We show: root + ellipsis + last-3. The ellipsis `Menu` lists the 4 hidden middle
segments, each navigable on click.

`breadcrumbs` is `internal` (not `private`) so `@testable import UIToolbar` tests can call it.

### `PaneToolbar.swift`

```swift
import Core
import DesignSystem
import SwiftUI

public struct PaneToolbar: View {
    @Bindable private var viewModel: PaneToolbarViewModel

    @Environment(\.theme) private var theme

    public init(viewModel: PaneToolbarViewModel) {
        self.viewModel = viewModel
    }

    public var body: some View {
        VStack(spacing: 0) {
            HStack(spacing: Spacing.sm) {
                navButtons
                Divider().frame(height: 16)
                viewModePicker
                Spacer()
                SearchField(debouncer: viewModel.searchDebouncer)
                    .frame(width: 180)
                actionButtons
            }
            .padding(.horizontal, Spacing.sm)
            .padding(.vertical, Spacing.xs)
            .background(theme.colors.surface)

            Divider()

            PathBar(
                path: viewModel.currentPath,
                onNavigate: { viewModel.navigate(to: $0) }
            )

            Divider()
        }
    }

    private var navButtons: some View {
        HStack(spacing: Spacing.xs) {
            toolbarButton(systemName: "chevron.left", disabled: !viewModel.canGoBack) {
                viewModel.goBack()
            }
            toolbarButton(systemName: "chevron.right", disabled: !viewModel.canGoForward) {
                viewModel.goForward()
            }
            toolbarButton(systemName: "arrow.up", disabled: !viewModel.canGoUp) {
                viewModel.goUp()
            }
        }
    }

    private var viewModePicker: some View {
        HStack(spacing: 0) {
            ForEach(ViewMode.allCases, id: \.self) { mode in
                Button { viewModel.setViewMode(mode) } label: {
                    Image(systemName: viewModeIcon(for: mode))
                        .foregroundStyle(
                            viewModel.viewMode == mode
                                ? theme.colors.accent
                                : theme.colors.textSecondary
                        )
                }
                .buttonStyle(.plain)
                .padding(Spacing.xs)
            }
        }
    }

    private var actionButtons: some View {
        HStack(spacing: Spacing.xs) {
            toolbarButton(systemName: "arrow.clockwise") { viewModel.refresh() }
            toolbarButton(systemName: "folder.badge.plus") { viewModel.newFolder() }
        }
    }

    private func toolbarButton(
        systemName: String,
        disabled: Bool = false,
        action: @escaping () -> Void
    ) -> some View {
        Button(action: action) {
            Image(systemName: systemName)
                .foregroundStyle(disabled ? theme.colors.textSecondary : theme.colors.textPrimary)
        }
        .buttonStyle(.plain)
        .disabled(disabled)
    }

    private func viewModeIcon(for mode: ViewMode) -> String {
        switch mode {
        case .list: "list.bullet"
        case .columns: "rectangle.split.3x1"
        case .icons: "square.grid.2x2"
        }
    }
}
```

**Rationale for extracted sub-views:** `navButtons`, `viewModePicker`, `actionButtons` keep
`body` simple and avoid SwiftLint `cyclomatic_complexity` warnings. Each sub-view helper is
a computed `some View` property, not a separate file (they're too tightly coupled to split).

**Why two `Task.yield()` calls suffice:** With `sleep: { _ in }`, the debounce Task suspends
once (to start), then the no-op sleep returns, then it fires the callback. Two yields give the
executor two rounds to advance the task. If tests are flaky, add a third yield.

## Key design decisions

| Decision | Rationale |
|----------|-----------|
| `HistoryStack` as pure `struct` | `Sendable`, no identity, trivially testable. Mutated by the `@MainActor` ViewModel which owns the struct. |
| Debounce in `SearchDebouncer`, not `SearchField` | ViewModel-layer classes are testable with XCTest; SwiftUI views are not. Injecting a mock sleep closure eliminates real-time dependency. |
| `SleepFunction = @Sendable (Duration) async throws -> Void` | `Duration` is a Swift stdlib type (macOS 13+, within our macOS 14 target). Matches `Task.sleep(for:)`. Simpler than injecting `any Clock<Swift.Duration>` (which requires a generic or complex existential on the `@Observable` class). |
| `PathSegmentData` and `BreadcrumbItem` at file scope | SwiftLint `nesting` rule rejects types nested more than 1 level deep. A `struct` inside an `enum` inside a file would be level 2 and fail `--strict`. |
| `PathBar.breadcrumbs` marked `internal` | Allows `@testable import UIToolbar` to call it directly in breadcrumb-derivation tests, avoiding the need to spin up a full SwiftUI host. |
| Static `maxVisible = 5` overflow threshold | Deterministic; avoids `GeometryReader` layout complexity. A layout-driven approach is post-MVP. |
| `setViewMode` gates columns/icons at ViewModel | The "not yet implemented" signal stays in the ViewModel contract; the shell decides how to surface it (alert, toast, or ignore). |
| `searchDebouncer` publicly exposed on ViewModel | Shell wires `vm.searchDebouncer.onFire` to filter the pane; no need for an additional delegation layer. |
| Package.swift adds `DesignSystem` to `UIToolbar` | Internal target dependency — permitted by the session-01 freeze rule. Recorded in handoff. |

## Risks and mitigations

| Risk | Mitigation |
|------|-----------|
| SwiftLint `nesting`: nested types in `BreadcrumbItem` | Fixed by placing `PathSegmentData` and `BreadcrumbItem` at file scope in `PathBarSegment.swift`. |
| SwiftLint `cyclomatic_complexity` in `PaneToolbar.body` | Extracted into `navButtons`, `viewModePicker`, `actionButtons` computed properties. |
| `@unchecked Sendable` in test support | `RecordingCallback` is not used — use a plain `var` captured by `inout` in test methods, or a local `class` with a justifying comment. |
| `Task.yield()` debounce test determinism | Two yields are sufficient with `sleep: { _ in }`. Documented in handoff; session implementer may add a third if flaky. |
| Package.swift outside touch glob | Necessary internal wiring exception. Document in handoff. |
| `@Bindable` availability | `@Bindable` is `@_alwaysEmitIntoClient` and available on macOS 14+ from `Observation` framework. Confirmed available on our deployment target. |
| `@MainActor (FilePath) async -> [FilePath]` closure type | `@MainActor` closures are implicitly `Sendable` in Swift 6 because the actor provides isolation. No explicit `@Sendable` annotation needed. |
| SwiftFormat `--lint` argument order | Paths before `--lint`: `swiftformat Sources/UI/Toolbar Tests/UITests/ToolbarTests --lint`. |
| Trailing commas | All array/call trailing commas required; SwiftLint `mandatory_comma: true` in effect. |

## Shell callback contract (for Session 26 handoff)

Session 26 (Main Window Shell) must wire the ViewModel before embedding `PaneToolbar`:

```swift
// Construct one ViewModel per pane (Session 26)
let vm = PaneToolbarViewModel(initialPath: startingPath)

// Required — shell performs actual FS navigation
vm.onNavigate = { [weak paneController] path in
    Task { await paneController?.enumerate(path) }
}
// Required — shell triggers directory re-enumeration
vm.onRefresh = { [weak paneController] in
    Task { await paneController?.refresh() }
}
// Required — shell shows new-folder dialog
vm.onNewFolder = { [weak paneController] in
    paneController?.showNewFolderSheet()
}
// Optional — shell may show a "not in MVP" alert
vm.onViewModeUnavailable = { mode in
    // No-op or show alert
}
// Search: wire debouncer to pane's filter
vm.searchDebouncer.onFire = { [weak paneController] term in
    paneController?.filterItems(matching: term)
}

// Embed:
PaneToolbar(viewModel: vm)

// CRITICAL: route all external navigations through vm.navigate(to:)
// so history stays in sync. Never update currentPath directly.
// Example — double-click in file pane:
vm.navigate(to: clickedItemPath)
```
