### 1.4 AppMetadataReader

**`Sources/Features/Uninstaller/AppMetadataReader.swift`**

```swift
import Foundation

public struct AppMetadataReader: AppMetadataReading {
    public init() {}

    public func readMetadata(from bundleURL: URL) throws -> AppMetadata {
        let plistURL = bundleURL.appending(path: "Contents/Info.plist")
        guard FileManager.default.fileExists(atPath: plistURL.path(percentEncoded: false)) else {
            throw UninstallerError.notAnAppBundle(bundleURL)
        }
        let data = try Data(contentsOf: plistURL)
        guard
            let plist = try PropertyListSerialization.propertyList(from: data, format: nil) as? [String: Any],
            let bundleID = plist["CFBundleIdentifier"] as? String,
            let executableName = plist["CFBundleExecutable"] as? String
        else {
            throw UninstallerError.invalidInfoPlist(bundleURL)
        }
        let bundleName = (plist["CFBundleName"] as? String)
            ?? (plist["CFBundleDisplayName"] as? String)
            ?? bundleURL.deletingPathExtension().lastPathComponent
        let version = plist["CFBundleShortVersionString"] as? String
        let size = computeBundleSize(bundleURL)
        return AppMetadata(
            bundleURL: bundleURL,
            bundleID: bundleID,
            bundleName: bundleName,
            executableName: executableName,
            version: version,
            bundleSizeInBytes: size
        )
    }

    private func computeBundleSize(_ url: URL) -> Int64 {
        let keys: Set<URLResourceKey> = [.fileSizeKey, .isDirectoryKey]
        guard let enumerator = FileManager.default.enumerator(
            at: url,
            includingPropertiesForKeys: Array(keys),
            options: [.skipsPackageDescendants]
        ) else { return 0 }
        var total: Int64 = 0
        for case let fileURL as URL in enumerator {
            guard let values = try? fileURL.resourceValues(forKeys: keys),
                  values.isDirectory != true
            else { continue }
            total += Int64(values.fileSize ?? 0)
        }
        return total
    }
}
```

### 1.5 MatchScorer

**`Sources/Features/Uninstaller/MatchScorer.swift`**

Scoring algorithm (returns score in 0.0…1.0 and a human-readable reason string):

1. Extract the last reverse-DNS component of bundleID (e.g., `"dropbox"` from `"com.getdropbox.dropbox"`). If this component (lowercased, ≥4 chars to avoid false positives on common words) appears in the candidate path → score 0.9, reason "Bundle ID match".
2. If bundleName (lowercased, ≥3 chars) appears in the candidate path (lowercased) → score 0.7, reason "App name match".
3. If executableName (lowercased, ≥3 chars) appears in candidate path (lowercased) → score 0.5, reason "Executable name match".
4. Otherwise score 0.1, reason "Partial match".

```swift
public enum MatchScorer {
    public static func score(url: URL, metadata: AppMetadata) -> (score: Double, reason: String) {
        let pathLowercased = url.path(percentEncoded: false).lowercased()
        
        // bundleID: last component (after last "."), minimum 4 chars
        let bundleComponents = metadata.bundleID.lowercased().split(separator: ".")
        if let lastComponent = bundleComponents.last,
           lastComponent.count >= 4,
           pathLowercased.contains(lastComponent) {
            return (0.9, "Bundle ID match")
        }
        
        let name = metadata.bundleName.lowercased()
        if name.count >= 3, pathLowercased.contains(name) {
            return (0.7, "App name match")
        }
        
        let exec = metadata.executableName.lowercased()
        if exec.count >= 3, pathLowercased.contains(exec) {
            return (0.5, "Executable name match")
        }
        
        return (0.1, "Partial match")
    }

    public static func confidence(url: URL, metadata: AppMetadata) -> (confidence: Confidence, reason: String) {
        let (score, reason) = Self.score(url: url, metadata: metadata)
        return (Confidence.from(score: score), reason)
    }
}
```

### 1.6 AssociatedFilesScanner

**`Sources/Features/Uninstaller/AssociatedFilesScanner.swift`**

`public actor AssociatedFilesScanner: AssociatedFilesScanning`

Algorithm:
1. Walk each path in `SearchPaths.userPaths` — enumerate top-level items (not recursive) using `FileManager.default.contentsOfDirectory(at:includingPropertiesForKeys:)`.
2. Walk each path in `SearchPaths.systemPaths` — same approach; mark `requiresAdmin = true`.
3. For each item URL, call `MatchScorer.confidence(url:metadata:)`.
4. Discard items with score < 0.1 (no match).
5. Read `sizeInBytes` and `lastModified` via `URLResourceValues`.
6. Return `[AssociatedFile]` sorted by confidence descending.

```swift
public actor AssociatedFilesScanner: AssociatedFilesScanning {
    public init() {}

    public func scan(for metadata: AppMetadata) async throws -> [AssociatedFile] {
        var results: [AssociatedFile] = []
        let resourceKeys: Set<URLResourceKey> = [.fileSizeKey, .contentModificationDateKey, .isDirectoryKey]

        for searchDir in SearchPaths.userPaths {
            results += scanDirectory(searchDir, metadata: metadata, requiresAdmin: false, keys: resourceKeys)
        }
        for searchDir in SearchPaths.systemPaths {
            results += scanDirectory(searchDir, metadata: metadata, requiresAdmin: true, keys: resourceKeys)
        }

        return results.sorted { $0.confidence > $1.confidence }
    }

    private func scanDirectory(
        _ directory: URL,
        metadata: AppMetadata,
        requiresAdmin: Bool,
        keys: Set<URLResourceKey>
    ) -> [AssociatedFile] {
        guard let items = try? FileManager.default.contentsOfDirectory(
            at: directory,
            includingPropertiesForKeys: Array(keys),
            options: [.skipsHiddenFiles]
        ) else { return [] }

        return items.compactMap { url in
            let (confidence, reason) = MatchScorer.confidence(url: url, metadata: metadata)
            guard confidence > .low || confidence == .low else {
                // include low confidence; exclude score-0.1 items
                return nil
            }
            let (score, _) = MatchScorer.score(url: url, metadata: metadata)
            guard score >= 0.1 else { return nil }
            let values = try? url.resourceValues(forKeys: keys)
            return AssociatedFile(
                url: url,
                sizeInBytes: Int64(values?.fileSize ?? 0),
                lastModified: values?.contentModificationDate ?? Date.distantPast,
                confidence: confidence,
                reason: reason,
                requiresAdmin: requiresAdmin
            )
        }
    }
}
```

> **Note on score threshold:** `score >= 0.1` means "any structural match at all". This intentionally includes low-confidence results so users can see them collapsed under "Show all". The UI default-selection rules filter high vs medium vs low for checked state.

### 1.7 UninstallExecutor

**`Sources/Features/Uninstaller/UninstallExecutor.swift`**

`public actor UninstallExecutor: UninstallExecuting`

Rules:
- Uses `FileManager.default.trashItem(at:resultingItemURL:)` exclusively. Never `removeItem`.
- Skips files where `requiresAdmin == true` (scanner flags these; executor enforces the skip).
- Trashes associated files first, then the app bundle last.
- Wraps `FileManager` errors in `UninstallerError.trashFailed`.

```swift
public actor UninstallExecutor: UninstallExecuting {
    public init() {}

    public func execute(plan: UninstallPlan) async throws {
        for file in plan.selectedFiles where !file.requiresAdmin {
            do {
                try FileManager.default.trashItem(at: file.url, resultingItemURL: nil)
            } catch {
                throw UninstallerError.trashFailed(file.url, error)
            }
        }
        do {
            try FileManager.default.trashItem(at: plan.appMetadata.bundleURL, resultingItemURL: nil)
        } catch {
            throw UninstallerError.trashFailed(plan.appMetadata.bundleURL, error)
        }
    }
}
```

### 3.2 UninstallerViewModel

**`Sources/UI/UninstallerUI/UninstallerViewModel.swift`**

Key design choices:
- `ScanState` is a top-level enum in `UninstallerViewModel.swift` (would violate nesting rule if nested, but since it's in the same file it's acceptable as a file-scope type — define it as `public enum ScanState` at file scope)
- Sorting is computed in ViewModel (not in View), making it unit-testable
- Default selection applied immediately when scan completes via `applyDefaultSelections`
- `@ObservationIgnored` on all injected protocols and internal tasks

```swift
import Core
import DesignSystem
import FeaturesUninstaller
import Foundation
import SwiftUI

public enum ScanState: Sendable {
    case idle
    case scanning
    case ready([AssociatedFile])
    case failed(String)
}

@MainActor
@Observable
public final class UninstallerViewModel {
    // MARK: - Observable state

    public private(set) var appMetadata: AppMetadata?
    public private(set) var scanState: ScanState = .idle
    public var selectedIDs: Set<UUID> = []
    public var showLowConfidence: Bool = false
    public var sortKey: AssociatedFileSortKey = .confidence
    public var sortAscending: Bool = true
    public private(set) var dropError: String?
    public private(set) var isExecuting: Bool = false

    // MARK: - Injected (not observed)

    @ObservationIgnored private let metadataReader: any AppMetadataReading
    @ObservationIgnored private let scanner: any AssociatedFilesScanning
    @ObservationIgnored private let executor: any UninstallExecuting
    @ObservationIgnored private var scanTask: Task<Void, Never>?

    // MARK: - Init

    public init(
        metadataReader: some AppMetadataReading,
        scanner: some AssociatedFilesScanning,
        executor: some UninstallExecuting
    ) {
        self.metadataReader = metadataReader
        self.scanner = scanner
        self.executor = executor
    }

    // MARK: - Computed state

    public var allFiles: [AssociatedFile] {
        guard case .ready(let files) = scanState else { return [] }
        return files
    }

    // Files shown in the table: high + medium always; low only when showLowConfidence
    public var displayedFiles: [AssociatedFile] {
        let visible = allFiles.filter { showLowConfidence || $0.confidence > .low }
        return sorted(visible)
    }

    public var lowConfidenceCount: Int {
        allFiles.filter { $0.confidence == .low }.count
    }

    public var selectedFiles: [AssociatedFile] {
        allFiles.filter { selectedIDs.contains($0.id) }
    }

    public var selectedItemCount: Int {
        selectedFiles.count + (appMetadata != nil ? 1 : 0)
    }

    public var totalSelectedBytes: Int64 {
        (appMetadata?.bundleSizeInBytes ?? 0) + selectedFiles.reduce(0) { $0 + $1.sizeInBytes }
    }

    public var confirmationText: String {
        let formatted = ByteCountFormatter.string(
            fromByteCount: totalSelectedBytes,
            countStyle: .file
        )
        let count = selectedItemCount
        let itemLabel = count == 1 ? "item" : "items"
        return "Move \(count) \(itemLabel) (\(formatted)) to Trash"
    }

    // MARK: - Actions

    public func load(appURL: URL) async {
        dropError = nil
        scanTask?.cancel()

        // Validate: must be .app with Contents/Info.plist
        guard appURL.pathExtension == "app",
              FileManager.default.fileExists(
                  atPath: appURL.appending(path: "Contents/Info.plist").path(percentEncoded: false)
              )
        else {
            dropError = "Drop a valid .app bundle to begin."
            return
        }

        scanState = .scanning
        selectedIDs = []
        appMetadata = nil

        scanTask = Task {
            do {
                let metadata = try metadataReader.readMetadata(from: appURL)
                self.appMetadata = metadata
                let files = try await scanner.scan(for: metadata)
                self.scanState = .ready(files)
                self.applyDefaultSelections(files: files)
            } catch {
                self.scanState = .failed(error.localizedDescription)
            }
        }
    }

    // High-confidence, non-admin files are checked by default
    func applyDefaultSelections(files: [AssociatedFile]) {
        selectedIDs = Set(
            files
                .filter { $0.confidence == .high && !$0.requiresAdmin }
                .map(\.id)
        )
    }

    public func toggleSelection(_ id: UUID) {
        guard let file = allFiles.first(where: { $0.id == id }),
              !file.requiresAdmin
        else { return }
        if selectedIDs.contains(id) {
            selectedIDs.remove(id)
        } else {
            selectedIDs.insert(id)
        }
    }

    public func confirmUninstall() async {
        guard let metadata = appMetadata else { return }
        isExecuting = true
        defer { isExecuting = false }
        let plan = UninstallPlan(appMetadata: metadata, selectedFiles: selectedFiles)
        try? await executor.execute(plan: plan)
    }

    public func cancel() {
        scanTask?.cancel()
        scanState = .idle
        appMetadata = nil
        selectedIDs = []
        dropError = nil
    }

    // MARK: - Sorting

    private func sorted(_ files: [AssociatedFile]) -> [AssociatedFile] {
        files.sorted { a, b in
            let result: Bool
            switch sortKey {
            case .path: result = a.url.lastPathComponent < b.url.lastPathComponent
            case .size: result = a.sizeInBytes < b.sizeInBytes
            case .modified: result = a.lastModified < b.lastModified
            case .confidence: result = a.confidence < b.confidence
            }
            return sortAscending ? result : !result
        }
    }

    deinit {
        scanTask?.cancel()
    }
}
```

### 3.4 AssociatedFilesTable

**`Sources/UI/UninstallerUI/AssociatedFilesTable.swift`**

Design: SwiftUI `Table` for sortable columns on the visible files (high + medium by default).
The `sortOrder` binding uses `[KeyPathComparator<AssociatedFile>]`.
Low-confidence items are toggled via a `Button` below the table.
System paths show a disabled `Toggle` with a lock icon.

```swift
import DesignSystem
import FeaturesUninstaller
import SwiftUI

public struct AssociatedFilesTable: View {
    @Binding var selectedIDs: Set<UUID>
    @Binding var showLowConfidence: Bool
    let files: [AssociatedFile]
    let lowConfidenceCount: Int
    let onToggle: (UUID) -> Void

    @State private var sortOrder: [KeyPathComparator<AssociatedFile>] = [
        KeyPathComparator(\.confidence, order: .reverse),
    ]

    private var displayedFiles: [AssociatedFile] {
        let visible = files.filter { showLowConfidence || $0.confidence > .low }
        return visible.sorted(using: sortOrder)
    }

    public var body: some View {
        VStack(spacing: 0) {
            Table(displayedFiles, sortOrder: $sortOrder) {
                // Checkbox column — no sortable header
                TableColumn("") { row in
                    Toggle("", isOn: Binding(
                        get: { selectedIDs.contains(row.id) },
                        set: { _ in onToggle(row.id) }
                    ))
                    .toggleStyle(.checkbox)
                    .disabled(row.requiresAdmin)
                    .overlay(alignment: .trailing) {
                        if row.requiresAdmin {
                            Image(systemName: "lock.fill")
                                .foregroundStyle(.secondary)
                                .font(.caption)
                        }
                    }
                }
                .width(32)

                TableColumn("Path", value: \.url.lastPathComponent) { row in
                    Text(redactedPath(row.url))
                        .font(.caption)
                        .lineLimit(1)
                        .truncationMode(.middle)
                }

                TableColumn("Size", value: \.sizeInBytes) { row in
                    Text(ByteCountFormatter.string(fromByteCount: row.sizeInBytes, countStyle: .file))
                        .font(.caption)
                }

                TableColumn("Modified", value: \.lastModified) { row in
                    Text(row.lastModified, style: .date)
                        .font(.caption)
                }

                TableColumn("Confidence", value: \.confidence) { row in
                    confidenceBadge(row.confidence)
                }

                TableColumn("Reason", value: \.reason) { row in
                    Text(row.reason)
                        .font(.caption)
                        .foregroundStyle(.secondary)
                }
            }
            .frame(minHeight: 200)

            if lowConfidenceCount > 0 {
                Divider()
                Button {
                    showLowConfidence.toggle()
                } label: {
                    Label(
                        showLowConfidence
                            ? "Hide low-confidence items"
                            : "Show \(lowConfidenceCount) low-confidence item\(lowConfidenceCount == 1 ? "" : "s")",
                        systemImage: showLowConfidence ? "chevron.up" : "chevron.down"
                    )
                    .font(.caption)
                }
                .buttonStyle(.plain)
                .padding(Spacing.sm)
            }
        }
    }

    private func redactedPath(_ url: URL) -> String {
        let home = FileManager.default.homeDirectoryForCurrentUser.path(percentEncoded: false)
        return url.path(percentEncoded: false).replacingOccurrences(of: home, with: "~")
    }

    @ViewBuilder
    private func confidenceBadge(_ confidence: Confidence) -> some View {
        let (label, color): (String, Color) = switch confidence {
        case .high: ("High", .green)
        case .medium: ("Medium", .orange)
        case .low: ("Low", .secondary)
        }
        Text(label)
            .font(.caption)
            .padding(.horizontal, 6)
            .padding(.vertical, 2)
            .background(color.opacity(0.15))
            .foregroundStyle(color)
            .clipShape(Capsule())
    }
}
```

> **`KeyPathComparator<AssociatedFile>`** requires `AssociatedFile` properties to support comparison.
> `url.lastPathComponent` is `String` (Comparable), `sizeInBytes` is `Int64` (Comparable),
> `lastModified` is `Date` (Comparable), `confidence` is `Confidence` (Comparable via rawValue),
> `reason` is `String` (Comparable). All columns are sortable natively.

## Design Decisions

| Decision | Rationale |
|---|---|
| Engine built in Phase 1 of Session 25 | Session 15 was never executed; `UIUninstallerUI` can't compile without it |
| `UninstallExecutor` is `actor` | FileManager calls are mutating side effects; actor serializes them safely |
| `AssociatedFilesScanner` is `actor` | Async scanning with cancellation support; isolates state |
| `AppMetadataReader` is `struct` | Pure synchronous function; no mutable state |
| Protocol fakes in `Sources/…/Testing/` | Production target, importable by both `FeaturesTests` and `UITests` |
| Default selection: high + non-admin only | Medium confidence is uncertain; user must opt in to avoid over-deletion |
| `requiresAdmin` disables toggle entirely | System paths must never be trashable without escalation; UI enforces this |
| Sorting in ViewModel, not View | Makes sort behavior unit-testable without rendering |
| `Table` with `KeyPathComparator` binding | Native macOS sort UX; no custom column header code needed |
| Low-confidence toggle below `Table` | `Table` has no native section/disclosure API; button below is clean |
| `ConfirmationFooter.onConfirm` calls `onDismiss` immediately | Async trash is fire-and-forget for MVP; transfer queue pattern can be added later |
| `NSWorkspace.shared.icon(forFile:)` called in `body` | `body` runs on main actor; safe without wrapping |
| `ByteCountFormatter.string(fromByteCount:countStyle:file)` | Matches macOS Finder display convention |

| Risk | Mitigation |
